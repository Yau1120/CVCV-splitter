# 2. 读取并解析 TXT
# ==========================================================

def parse_file(filepath):
    """
    读取两个部分：
    辅音 -> 元音，例如 Cp = {i, e, ə, u, a}
    元音 -> 辅音，例如 Ci = {p, pʰ, b, ...}

    返回两个 CV 组合集合及音素顺序。
    """
    text = filepath.read_text(encoding="utf-8-sig")

    forward = defaultdict(set)
    reverse = defaultdict(set)

    consonants = []
    vowels = []
    section = None

    for line in text.splitlines():
        line = line.strip()

        if not line:
            continue

        # 根据标题判断当前部分
        if "辅音" in line and "元音" in line and "→" in line:
            if line.index("辅音") < line.index("元音"):
                section = "forward"
            else:
                section = "reverse"
            continue

        match = re.match(r"^C(.+?)\s*=\s*\{(.*?)\}", line)

        if not match:
            continue

        symbol = match.group(1).strip()
        items = [
            item.strip()
            for item in match.group(2).split(",")
            if item.strip()
        ]

        if section == "forward":
            consonant = symbol

            if consonant not in consonants:
                consonants.append(consonant)

            for vowel in items:
                if vowel not in vowels:
                    vowels.append(vowel)
                forward[consonant].add(vowel)

        elif section == "reverse":
            vowel = symbol

            if vowel not in vowels:
                vowels.append(vowel)

            for consonant in items:
                if consonant not in consonants:
                    consonants.append(consonant)
                reverse[vowel].add(consonant)

    forward_pairs = {
        (c, v)
        for c, vowel_set in forward.items()
        for v in vowel_set
    }

    reverse_pairs = {
        (c, v)
        for v, consonant_set in reverse.items()
        for c in consonant_set
    }

    return (
        forward_pairs,
        reverse_pairs,
        consonants,
        vowels,
    )


# ==========================================================
# 3. 读取四个方言并整理数据
# ==========================================================

data = {}
all_consonants = []
all_vowels = []
discrepancies = []

for dialect, filepath in files.items():
    if not filepath.exists():
        raise FileNotFoundError(
            f"找不到文件：{filepath}\n"
            "请检查文件名及文件夹路径。"
        )

    forward_pairs, reverse_pairs, consonants, vowels = (
        parse_file(filepath)
    )

    # 两个方向不一致的组合
    for c, v in sorted(forward_pairs - reverse_pairs):
        discrepancies.append([
            dialect, c, v, "仅出现在辅音→元音部分"
        ])

    for c, v in sorted(reverse_pairs - forward_pairs):
        discrepancies.append([
            dialect, c, v, "仅出现在元音→辅音部分"
        ])

    # 取并集：任一方向记录的组合均保留
    data[dialect] = forward_pairs | reverse_pairs

    for c in consonants:
        if c not in all_consonants:
            all_consonants.append(c)

    for v in vowels:
        if v not in all_vowels:
            all_vowels.append(v)

all_consonants = list(dict.fromkeys(all_consonants))
all_vowels = list(dict.fromkeys(all_vowels))

if not all_consonants or not all_vowels:
    raise ValueError(
        "没有成功读取到辅音或元音。"
        "请检查 TXT 文件中的分节标题和 Cp = {...} 格式。"
    )


# ==========================================================
# 4. 通用函数：创建 CV 矩阵
# ==========================================================

def make_rich_plus(dialects):
    """创建同一单元格中不同颜色的 +。"""
    rich = CellRichText()

    for dialect in dialects:
        rich.append(
            TextBlock(
                InlineFont(
                    color=colors[dialect],
                    b=True,
                    sz=12,
                ),
                "+ ",
            )
        )

    return rich


def write_matrix(ws, dialects):
    """
    创建矩阵。
    dialects 决定本表显示哪些方言。
    """
    ws.cell(row=1, column=1, value="辅音 \\ 元音")

    # 元音表头
    for col, vowel in enumerate(all_vowels, start=2):
        cell = ws.cell(row=1, column=col, value=vowel)
        cell.font = Font(bold=True)
        cell.fill = header_fill
        cell.alignment = center
        cell.border = cell_border

    # 左上角
    cell = ws.cell(row=1, column=1)
    cell.font = Font(bold=True)
    cell.fill = header_fill
    cell.alignment = center
    cell.border = cell_border

    # 辅音行
    for row, consonant in enumerate(all_consonants, start=2):
        cell = ws.cell(row=row, column=1, value=consonant)
        cell.font = Font(bold=True)
        cell.fill = header_fill
        cell.alignment = center
        cell.border = cell_border

        # CV 组合
        for col, vowel in enumerate(all_vowels, start=2):
            cell = ws.cell(row=row, column=col)
            cell.alignment = center
            cell.border = cell_border

            present = [
                d for d in dialects
                if (consonant, vowel) in data[d]
            ]

            if present:
                cell.value = make_rich_plus(present)

    # 冻结表头与辅音列
    ws.freeze_panes = "B2"
    ws.column_dimensions["A"].width = 17

    for col in range(2, len(all_vowels) + 2):
        ws.column_dimensions[
            get_column_letter(col)
        ].width = 12

    for row in range(1, len(all_consonants) + 2):
        ws.row_dimensions[row].height = 24

    # 右侧图例
    legend_col = len(all_vowels) + 3

    cell = ws.cell(
        row=1, column=legend_col, value="方言图例"
    )
    cell.font = Font(bold=True)
    cell.alignment = center

    for i, dialect in enumerate(dialects, start=2):
        cell = ws.cell(
            row=i,
            column=legend_col,
            value=f"+  {dialect}",
        )
        cell.font = Font(
            color=colors[dialect],
            bold=True,
            size=12,
        )
        cell.alignment = center

    ws.column_dimensions[
        get_column_letter(legend_col)
    ].width = 16


# ==========================================================
# 5. 创建工作簿和四方言总表
# ==========================================================

wb = Workbook()

ws = wb.active

write_matrix(ws, list(files.keys()))
